Connect to Google Sheets

This app reads and writes directly to your TaskTracker spreadsheet via a Google Apps Script web app you deploy once.

  1. Open your Google Sheet → Extensions → Apps Script
  2. Replace all code with the script below, then save (Ctrl+S)
  3. Click Deploy → New deployment → type: Web app
  4. Set "Execute as" = Me, "Who has access" = Anyone
  5. Click Deploy, authorise, then copy the web app URL
  6. Paste it below and click Save

const SHEET_NAME = "TaskTracker";
const BALANCE_TOTAL = 20;

function doGet(e) {
  const action = e.parameter.action || "read";
  if (action === "read") return readData();
  return ContentService.createTextOutput(
    JSON.stringify({ error: "Unknown action" })
  ).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const action = data.action;
  if (action === "add")    return addRow(data.entry);
  if (action === "delete") return deleteRow(data.id);
  return ContentService.createTextOutput(
    JSON.stringify({ error: "Unknown action" })
  ).setMimeType(ContentService.MimeType.JSON);
}

function readData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  const rows = sheet.getDataRange().getValues();
  const headers = rows[0];
  const entries = rows.slice(1).map(r => ({
    id:      r[0],
    date:    r[1] ? Utilities.formatDate(new Date(r[1]), Session.getScriptTimeZone(), "yyyy-MM-dd") : "",
    balance: r[2],
    spent:   r[3],
    applied: r[4],
    remained:r[5],
    reason:  r[6],
    remark:  r[7],
    appDate: r[8] ? Utilities.formatDate(new Date(r[8]), Session.getScriptTimeZone(), "yyyy-MM-dd") : "",
    isBF:    r[0] === 0
  }));
  return respond({ entries, balance: BALANCE_TOTAL });
}

function addRow(entry) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  const rows = sheet.getDataRange().getValues();
  const dataRows = rows.slice(1).filter(r => r[0] !== 0);
  const nextId = dataRows.length > 0 ? Math.max(...dataRows.map(r => Number(r[0]))) + 1 : 1;
  const totalSpent = dataRows.reduce((s, r) => s + Number(r[4]), 0) + Number(entry.applied);
  const remained = BALANCE_TOTAL - parseFloat(totalSpent.toFixed(1));
  sheet.appendRow([
    nextId,
    entry.date,
    BALANCE_TOTAL,
    parseFloat(totalSpent.toFixed(1)),
    Number(entry.applied),
    parseFloat(remained.toFixed(1)),
    entry.reason,
    entry.remark,
    entry.appDate
  ]);
  updateFilter_();
  return respond({ ok: true, id: nextId });
}

function deleteRow(id) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  const rows = sheet.getDataRange().getValues();
  for (let i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][0]) === String(id)) {
      sheet.deleteRow(i + 1);
      break;
    }
  }
  recalcSpent_(sheet);
  return respond({ ok: true });
}

function recalcSpent_(sheet) {
  const rows = sheet.getDataRange().getValues();
  let spent = 0;
  for (let i = 1; i < rows.length; i++) {
    if (rows[i][0] === 0) continue;
    spent = parseFloat((spent + Number(rows[i][4])).toFixed(1));
    sheet.getRange(i + 1, 4).setValue(spent);
    sheet.getRange(i + 1, 6).setValue(parseFloat((BALANCE_TOTAL - spent).toFixed(1)));
  }
}

function updateFilter_() {
  try {
    const ss = SpreadsheetApp.getActive();
    const criteria = SpreadsheetApp.newFilterCriteria()
      .setHiddenValues(['', 'No']).build();
    ss.getActiveSheet().getFilter()
      .setColumnFilterCriteria(5, criteria);
  } catch(e) {}
}

function respond(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
